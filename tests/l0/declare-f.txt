until ! echo <(ls) |& true | g() { ! echo $(a

	plain
	E2

cat ${v:-w} $((1+2)) <<'E3'

f1 () 
{ 
    f b ~/d 'q r' & f a;
    until ! echo <(ls) 2>&1 | true | function g () 
    { 
        ! echo $(a
b) 't

b) && while time A=1 f a\ b a\ b;

plain
E2

        cat ${v:-w} $((1+2)) <<'E3'
two
lines
E3
